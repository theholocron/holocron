## Installation

```bash
pnpm add @theholocron/misc-utils
```

## Usage

```ts
import { errorMessage, konami } from "@theholocron/misc-utils";

try {
  doSomething();
} catch (err) {
  console.error(errorMessage(err));
}

document.addEventListener("keydown", (event) => {
  if (konami.is(event)) {
    console.log("Konami Code entered!");
  }
});
```
